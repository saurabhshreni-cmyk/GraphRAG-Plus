# Dependencies for the deployed backend (Vercel Python runtime).
#
# Vercel's Python functions get a 500 MB uncompressed bundle, 300 s max
# duration and 2 GB RAM on Hobby. That is enough for everything here EXCEPT
# PyTorch + sentence-transformers, so embeddings come from a hosted API
# instead (see graphrag_plus/app/embeddings/remote_embedder.py). Everything
# else - BM25, TF-IDF, FAISS, spaCy NER, the knowledge graph - runs in-process
# exactly as it does locally.
#
# Keep this in sync with the runtime deps in graphrag_plus/pyproject.toml.
# Anything added here ships; anything missing here breaks the deploy at import
# time, which is how /ingest/file silently 500'd every route before
# python-multipart was declared.

# --- web ---
fastapi>=0.111.0
# REQUIRED, not optional: main.py declares File()/Form() parameters, and
# FastAPI calls ensure_multipart_is_installed() at route-registration time -
# i.e. on import. Without this the app raises RuntimeError before the ASGI
# object exists and EVERY endpoint returns FUNCTION_INVOCATION_FAILED.
python-multipart>=0.0.9

# --- core ---
pydantic>=2.7.0
pydantic-settings>=2.2.1
networkx>=3.2.1
numpy>=1.26.4
rank-bm25>=0.2.2
python-dateutil>=2.9.0.post0
httpx>=0.27.0
prometheus-client>=0.20.0

# --- document ingestion ---
pypdf>=4.2.0
python-docx>=1.1.0
beautifulsoup4>=4.12.3

# --- durable corpus storage (Neon/Supabase Postgres) ---
# Optional at runtime: the app falls back to file storage under /tmp when
# GRAPHRAG_DATABASE_URL is unset.
psycopg[binary]>=3.1.18
